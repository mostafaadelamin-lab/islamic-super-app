# رتّل وصلِّ (rattilwasalli.com) on Bluehost / any Apache host.
# Replaces netlify.toml + the Netlify Function: static site from this folder, video API from api/videos.php.

Options -Indexes -MultiViews
DirectoryIndex index.html index.php
AddDefaultCharset utf-8
ErrorDocument 404 /404.html

<IfModule mod_rewrite.c>
  RewriteEngine On

  # Always HTTPS, and one address: rattilwasalli.com (www. goes to the bare domain).
  RewriteCond %{HTTPS} !=on
  RewriteCond %{HTTP:X-Forwarded-Proto} !=https
  RewriteRule ^ https://rattilwasalli.com%{REQUEST_URI} [L,R=301]
  RewriteCond %{HTTP_HOST} ^www\. [NC]
  RewriteRule ^ https://rattilwasalli.com%{REQUEST_URI} [L,R=301]

  # Video API (same paths the website already calls on Netlify).
  RewriteRule ^api/v1/reels/feed/?$ api/videos.php?route=feed [L,QSA]
  RewriteRule ^api/videos/search/?$ api/videos.php?route=search [L,QSA]
  RewriteRule ^api/v1/reels/channel/?$ api/videos.php?route=channel [L,QSA]
  RewriteRule ^api/channels/search/?$ api/videos.php?route=channels [L,QSA]

  # A page's data file (.../index.txt) opened as a page shows raw text (Next.js does this after an
  # upload while an older page is open). Browsers asking for a page get the page itself instead.
  RewriteCond %{HTTP:Sec-Fetch-Dest} =document [OR]
  RewriteCond %{HTTP_ACCEPT} text/html
  RewriteRule ^(.*/)?index\.txt$ /$1 [L,R=302]

  # Pages are folders with index.html (Next.js trailingSlash): /ar/help -> /ar/help/
  RewriteCond %{REQUEST_FILENAME} !-f
  RewriteCond %{REQUEST_FILENAME}/index.html -f
  RewriteRule ^(.*[^/])$ https://%{HTTP_HOST}/$1/ [L,R=301]
</IfModule>

# Types the PWA needs.
AddType application/manifest+json .webmanifest
AddType application/json .json
AddType text/javascript .js .mjs
AddType font/woff2 .woff2
AddType audio/mpeg .mp3
# Whole tafsir books (1.2.15): the app downloads data/tafsir/<book>/book.v1.dat for offline reading.
AddType application/octet-stream .dat

<IfModule mod_headers.c>
  Header always set X-Content-Type-Options "nosniff"
  Header always set Referrer-Policy "strict-origin-when-cross-origin"
  Header always set X-Frame-Options "SAMEORIGIN"
  # Bluehost's server adds "Permissions-Policy: geolocation=()", which blocks "use my location" (prayer times,
  # qibla, mosques). Replace it: location for this site only; microphone/camera stay off.
  Header always unset Permissions-Policy
  Header unset Permissions-Policy
  Header always set Permissions-Policy "geolocation=(self), microphone=(), camera=()"
  Header always set Strict-Transport-Security "max-age=31536000" env=HTTPS
  # Next.js hashed files never change: cache for a year.
  <If "%{REQUEST_URI} =~ m#^/_next/static/#">
    Header set Cache-Control "public, max-age=31536000, immutable"
  </If>
  # Pages, the service worker and the manifest must always be checked again, so updates show at once.
  <FilesMatch "(\.html|\.txt|sw\.js|\.webmanifest|^(lessons|sharawi|shabib)[^/]*\.json)$">
    Header set Cache-Control "no-cache, must-revalidate"
    Header unset Expires
  </FilesMatch>
  # Written by the upload tool while an upload runs: never cached anywhere.
  <Files "site-status.json">
    Header set Cache-Control "no-store"
    Header unset Expires
  </Files>
  # Page data files differ from the page itself: never mix them up in a cache.
  <FilesMatch "\.txt$">
    Header merge Vary "Accept, RSC, Next-Router-State-Tree, Next-Router-Prefetch"
  </FilesMatch>
</IfModule>

<IfModule mod_deflate.c>
  AddOutputFilterByType DEFLATE text/html text/plain text/css text/javascript application/javascript application/json application/manifest+json image/svg+xml
</IfModule>

# Netlify-only files, if they were uploaded by mistake.
<FilesMatch "^(netlify\.toml|.*\.mjs|.*\.d\.mts)$">
  Require all denied
</FilesMatch>
